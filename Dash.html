function renderTop5(data) { const extras = getExtras(data); let top5 = []; const el = document.getElementById("top5Box"); if (!el) return; // 1) Highest priority: latest SELECTED event const parsedFromSelected = extractTop5FromSelectedEvent(data); if (parsedFromSelected.length) { top5 = parsedFromSelected; } // 2) Fallback: extras.shortlist if (!top5.length && Array.isArray(extras.shortlist) && extras.shortlist.length) { top5 = extras.shortlist; } // 3) Fallback: subscribed_topk event (symbols only) if (!top5.length) { const events = getEvents(data); const subscribedEvt = [...events].reverse().find(ev => { const d = ev.details || ev.data || {}; return d.msg === "subscribed_topk" && Array.isArray(d.symbols); }); if (subscribedEvt) { const d = subscribedEvt.details || subscribedEvt.data || {}; top5 = d.symbols.map(sym => ({ symbol: sym })); } } if (!top5.length) { el.innerHTML = `
No top-5 shortlist available for this bot.
`; return; } const rows = top5.map((item, idx) => ` ${idx + 1} ${fmtText(item.symbol)} ${fmtText(item.bias || "--")} ${fmtNum(item.ltp)} ${fmtNum(item.pct)} ${fmtText(item.vol)} ${fmtNum(item.score)} `).join(""); el.innerHTML = ` ${rows}
# Symbol Bias LTP % Move Volume Score
`; }
